      const level = safeInfo("vocabularyLevel");
      ui.classDetails.replaceChildren(element("p", "", `族語別：${language}　／　方言別：${dialect}`), element("p", "", `授課對象：${audience}`));
      ui.teacherMeta.replaceChildren(element("p", "", `族語別：${language}　／　方言別：${dialect}`), element("p", "", `授課對象：${audience}　／　練習主題：${topic}`));
      ui.topicEyebrow.textContent = `${topic}主題${level ? ` · 詞彙難度：${level}` : ""}`;
    }

    // Fisher–Yates 洗牌。牌面不以字詞相似程度配對，只認資料 id 與不同卡牌種類。
    function shuffle(items) {
      const array = [...items];
      for (let i = array.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [array[i], array[j]] = [array[j], array[i]];
      }
      return array;
    }

    function resetRound(startImmediately) {
      state.session += 1;
      stopTimer();
      stopRecording();
      closeAllDialogs();
      state.phase = "idle";
      state.selected = [];
      state.matchedIds = new Set();
      state.matchedOrder = [];
      state.flips = 0;
      state.elapsedMs = 0;
      state.startedAt = null;
      state.feedbackEntry = null;
      state.feedbackMatched = false;
      state.lastIndex = 0;
      const entries = bank.filter((entry) => selectedIds.includes(entry.id));
      if (entries.length !== PAIR_COUNT) {
        ui.turnStatus.textContent = "請先選好 8 組完整詞彙。";
        return;
      }
      // 準備畫面也只顯示牌背；真正按下開始時再洗牌一次。
      state.cards = shuffle(entries.flatMap((entry) => [
        { pairId: entry.id, kind: "word", text: entry.word, entry },
        { pairId: entry.id, kind: "translation", text: entry.translation, entry }
      ]));
      renderBoard();
      renderStats();
      renderCollection();
      ui.readyShade.hidden = false;
      ui.restartBtn.hidden = false;
      ui.restartBtn.disabled = true;
      ui.resultBtn.hidden = true;
      ui.turnStatus.textContent = "按「開始遊戲」後再翻牌。";
      if (startImmediately) startRound();
    }

    function startRound() {
      if (state.phase !== "idle" || state.cards.length !== PAIR_COUNT * 2) return;
      unlockSuccessAudio(); // 只在學生的開始／重玩按鈕事件中建立 AudioContext。
      state.cards = shuffle(state.cards);
      state.phase = "playing";
      state.startedAt = performance.now();
      state.elapsedMs = 0;
      renderBoard();
      ui.readyShade.hidden = true;
      ui.restartBtn.disabled = false;
      ui.turnStatus.textContent = "先翻一張，再找它的另一半。";
      state.timerId = window.setInterval(renderTimer, 250);
      renderTimer();
      focusCard(0);
    }

    function stopTimer() {
      if (state.timerId !== null) window.clearInterval(state.timerId);
      state.timerId = null;
    }
    function elapsed() {
      return state.startedAt === null ? state.elapsedMs : Math.max(0, performance.now() - state.startedAt);
    }
    function formatTime(ms) {
      const totalSeconds = Math.floor(ms / 1000);
      const minutes = Math.floor(totalSeconds / 60);
      return `${String(minutes).padStart(2, "0")}:${String(totalSeconds % 60).padStart(2, "0")}`;
    }
    function renderTimer() {
      const ms = elapsed();
      const value = formatTime(ms);
      if (ui.timer.textContent !== value) ui.timer.textContent = value;
      const seconds = Math.floor(ms / 1000);
      ui.timer.setAttribute("aria-label", `經過時間 ${Math.floor(seconds / 60)} 分 ${seconds % 60} 秒`);
    }
    function renderStats() {
      ui.matchedCount.textContent = String(state.matchedIds.size);
      ui.flipCount.textContent = String(state.flips);
      renderTimer();
    }

    function renderBoard() {
      const fragment = document.createDocumentFragment();
      state.cards.forEach((card, index) => {
        const button = element("button", "card");
        button.type = "button";
        button.dataset.index = String(index);
        button.dataset.pairId = card.pairId;
        button.dataset.kind = card.kind;
        button.addEventListener("click", (event) => {
          if (event.detail > 1) return;
          selectCard(index);
        });
        fragment.appendChild(button);
      });
      ui.board.replaceChildren(fragment);
      updateAllCards();
    }
    function updateAllCards() { state.cards.forEach((_, index) => updateCard(index)); }
    function updateCard(index) {
      const card = state.cards[index];
      const button = ui.board.children[index];
      const matched = state.matchedIds.has(card.pairId);
      const up = matched || state.selected.includes(index);
      const blocked = state.phase !== "playing" || up;
      button.disabled = state.phase === "idle";
      button.classList.toggle("is-up", up);
      button.classList.toggle("is-matched", matched);
      button.setAttribute("aria-pressed", String(up));
      button.setAttribute("aria-disabled", String(blocked));
      button.tabIndex = matched ? -1 : 0;
      button.replaceChildren();
      if (up) {
        const face = element("span", "face");
        face.setAttribute("aria-hidden", "true");
        face.append(element("span", "card-kind", card.kind === "word" ? "族語" : "中文"));
        const word = element("span", "card-word", card.text);
        word.dir = "auto";
        face.append(word);
        button.append(face);
        if (matched) { const mark = element("span", "card-mark", "✓"); mark.setAttribute("aria-hidden", "true"); button.append(mark); }
        button.setAttribute("aria-label", `第 ${index + 1} 張，${card.kind === "word" ? "族語" : "中文"}：${card.text}，${matched ? "已配對完成" : "已翻開"}`);
      } else {
        // 蓋牌時不把答案放在可讀文字、title 或 aria-label 中。
        const dots = element("span", "dots");
        dots.setAttribute("aria-hidden", "true");
        for (let n = 0; n < 4; n++) dots.append(element("i"));
        button.append(dots);
        button.setAttribute("aria-label", `第 ${index + 1} 張，尚未翻開`);
      }
    }

    function selectCard(index) {
      const card = state.cards[index];
      if (!card || state.phase !== "playing" || state.selected.length >= 2 || state.selected.includes(index) || state.matchedIds.has(card.pairId)) return;
      state.lastIndex = index;
      state.selected.push(index);
      state.flips += 1;
      updateCard(index);
      renderStats();
      if (state.selected.length === 1) {
        ui.turnStatus.textContent = "已翻開一張，請再選另一張。";
        return;
      }
      // 同步上鎖，不等動畫、音檔或 setTimeout，第三張不可能插入比對。
      state.phase = "feedback";
      const [first, second] = state.selected.map((i) => state.cards[i]);
      const isMatch = first.pairId === second.pairId && first.kind !== second.kind;
      state.feedbackMatched = isMatch;
      state.feedbackEntry = isMatch ? first.entry : null;
      if (isMatch) {
        state.matchedIds.add(first.pairId);
        state.matchedOrder.push(first.entry);
        if (state.matchedIds.size === PAIR_COUNT) {
          state.elapsedMs = elapsed();
          state.startedAt = null;
          stopTimer();
        }
        renderCollection();
        playSuccessTone();
      }
      updateAllCards();
      renderStats();
      openFeedback(isMatch, first, second);
    }

    function renderCollection() {
      const entries = state.matchedOrder;
      const fragment = document.createDocumentFragment();
      for (let index = 0; index < PAIR_COUNT; index++) {
        const entry = entries[index];
        const item = element("li", "collection-item");
        const number = element("span", "item-number", String(index + 1).padStart(2, "0"));
        number.setAttribute("aria-hidden", "true");
        item.append(number);
        if (entry) {
          const content = element("span", "item-content");
          const word = element("span", "item-word", entry.word); word.dir = "auto";
          const translation = element("span", "item-translation", entry.translation); translation.dir = "auto";
          content.append(word, translation); item.append(content);
          const check = element("span", "item-check", "✓"); check.setAttribute("aria-hidden", "true"); item.append(check);
        } else {
          const line = element("span", "item-empty"); line.setAttribute("aria-hidden", "true"); item.append(line, element("span", "sr-only", "尚未採集"));
        }
        fragment.append(item);
      }
      ui.collectionList.replaceChildren(fragment);
      const marks = [];
      for (let index = 0; index < PAIR_COUNT; index++) marks.push(element("span", index < entries.length ? "done" : ""));
      ui.progressTrack.replaceChildren(...marks);
      ui.collectionCaption.textContent = entries.length === PAIR_COUNT ? "這一回合的 8 組詞彙，都收集好了。" : entries.length ? `已留下 ${entries.length} 組詞彙，再慢慢找下一組。` : "配對成功的詞彙會留在這裡。";
    }

    function openFeedback(isMatch, first, second) {
      stopRecording();
      ui.audioStatus.textContent = "";
      ui.pronounceLabel.textContent = "聽發音";
      ui.successBadge.hidden = !isMatch;
      ui.successContent.hidden = !isMatch;
      ui.mismatchContent.hidden = isMatch;
      ui.feedbackTitle.textContent = isMatch ? "找到一組！" : "還不是同一組";
      ui.feedbackIntro.textContent = isMatch ? "讀一讀，再把這個詞收進記憶。" : "先記住這兩張的位置，準備好再蓋回。";
      ui.feedbackContinue.textContent = isMatch ? "繼續" : "再試一次";
      ui.exampleSection.hidden = true;
      ui.audioSection.hidden = true;
      ui.exampleText.textContent = "";
      if (isMatch) {
        const entry = first.entry;
        ui.successProgress.textContent = `已完成 ${state.matchedIds.size} / 8 組`;
        ui.matchedWord.textContent = entry.word;
        ui.matchedTranslation.textContent = entry.translation;
        if (entry.exampleSentence.trim()) { ui.exampleSection.hidden = false; ui.exampleText.textContent = entry.exampleSentence; }
        if (entry.audioUrl) ui.audioSection.hidden = false;
        ui.turnStatus.textContent = "配對成功，讀完後按「繼續」。";
      } else {
        const peeks = [first, second].map((card, i) => {
          const box = element("div", "memory-peek");
          box.append(element("span", "card-kind", `第 ${state.selected[i] + 1} 張 · ${card.kind === "word" ? "族語" : "中文"}`));
          const word = element("p", "card-word", card.text); word.dir = "auto"; box.append(word);
          return box;
        });
        ui.mismatchCards.replaceChildren(...peeks);
        ui.turnStatus.textContent = "還沒配對，記住位置後按「再試一次」。";
      }
      showDialog(ui.feedbackDialog, ui.feedbackTitle);
    }

    function continueAfterFeedback() {
      if (state.phase !== "feedback") return;
      const lastIndex = state.lastIndex;
      stopRecording();
      hideDialog(ui.feedbackDialog, false);
      state.selected = [];
      state.feedbackEntry = null;
      if (state.matchedIds.size === PAIR_COUNT) {
        state.phase = "finished";
        updateAllCards();
        ui.turnStatus.textContent = "這一回合，全部收藏了。";
        ui.restartBtn.hidden = true;
        ui.resultBtn.hidden = false;
        showResults();
        return;
      }
      state.phase = "playing";
      updateAllCards();
      ui.turnStatus.textContent = state.feedbackMatched ? "收下一組記憶，再找下一組。" : "沒關係，再翻兩張試試看。";
      focusCard(nextPlayableIndex(lastIndex, 1));
    }

    function showResults() {
      ui.resultFlips.textContent = String(state.flips);
      ui.resultTime.textContent = formatTime(state.elapsedMs);
      showDialog(ui.resultDialog, ui.resultTitle);
    }

    function focusCard(index) {
      const button = ui.board.children[index];
      if (button && !button.disabled) button.focus({ preventScroll: true });
    }
    function nextPlayableIndex(from, step) {
      const count = state.cards.length;
      if (!count) return 0;
      let index = from;
      for (let tried = 0; tried < count; tried++) {
        index = (index + step + count) % count;
        const card = state.cards[index];
        if (!state.matchedIds.has(card.pairId) && !state.selected.includes(index)) return index;
      }
      return from;
    }
    ui.board.addEventListener("keydown", (event) => {
      const target = event.target.closest("button.card");
      if (!target || state.phase !== "playing") return;
      const index = Number(target.dataset.index);
      const columns = getComputedStyle(ui.board).gridTemplateColumns.split(" ").filter(Boolean).length;
      const step = { ArrowRight: 1, ArrowLeft: -1, ArrowDown: columns, ArrowUp: -columns }[event.key];
      if (step !== undefined) {
        event.preventDefault();
        const next = nextPlayableIndex(index, step);
        // 同欄均已完成時改找下一張，避免方向鍵卡住。
        focusCard(next === index ? nextPlayableIndex(index, Math.sign(step)) : next);
      } else if (event.key === "Home") {
        event.preventDefault(); focusCard(nextPlayableIndex(state.cards.length - 1, 1));
      } else if (event.key === "End") {
        event.preventDefault(); focusCard(nextPlayableIndex(0, -1));
      }
    });
    // 長按 Enter／空白鍵不連續觸發按鈕；有效操作仍使用原生按鈕鍵盤行為。
    document.addEventListener("keydown", (event) => {
      if (event.repeat && (event.key === "Enter" || event.key === " ") && event.target.closest("button")) {
        event.preventDefault(); event.stopImmediatePropagation();
      }
    }, true);

    function showDialog(dialog, initialFocus) {
      focusReturn = document.activeElement;
      if (!dialog.open) dialog.showModal();
      document.body.classList.add("has-dialog");
      if (initialFocus) initialFocus.focus({ preventScroll: true });
      dialog.scrollTop = 0;
    }
    function hideDialog(dialog, restore = true) {
      if (dialog.open) dialog.close();
      if (!document.querySelector("dialog[open]")) document.body.classList.remove("has-dialog");
      if (restore && focusReturn && focusReturn.isConnected && !focusReturn.disabled) focusReturn.focus({ preventScroll: true });
    }
    function closeAllDialogs() {
      document.querySelectorAll("dialog[open]").forEach((dialog) => dialog.close());
      document.body.classList.remove("has-dialog");
    }
    document.querySelectorAll("dialog").forEach((dialog) => {
      dialog.addEventListener("close", () => {
        if (!document.querySelector("dialog[open]")) document.body.classList.remove("has-dialog");
      });
    });
    ui.feedbackDialog.addEventListener("cancel", (event) => {
      event.preventDefault(); // 不用 Esc 或點背景跳過學生主動確認。
      ui.feedbackContinue.focus();
    });
    ui.resultDialog.addEventListener("cancel", () => {
      window.setTimeout(() => ui.resultBtn.focus({ preventScroll: true }), 0);
    });

    function openTeacher() {
      if (!bank.length || state.phase === "feedback") return;
      draftIds = new Set(selectedIds);
      renderChoices();
      showDialog(ui.teacherDialog, ui.teacherTitle);
    }
    function renderChoices() {
      const fragment = document.createDocumentFragment();
      const legend = element("legend", "sr-only", "本局的 8 組詞彙"); fragment.append(legend);
      bank.forEach((entry) => {
        const label = element("label", "word-choice");
        const input = document.createElement("input");
        input.type = "checkbox"; input.value = entry.id; input.checked = draftIds.has(entry.id);
        input.addEventListener("change", () => {
          if (input.checked) draftIds.add(entry.id); else draftIds.delete(entry.id);
          updateChoiceControls();
        });
        const content = element("span", "choice-content");
        const word = element("span", "choice-word", entry.word); word.dir = "auto";
        content.append(word, element("span", "choice-translation", entry.translation));
        label.append(input, content); fragment.append(label);
      });
      ui.wordChoices.replaceChildren(fragment);
      updateChoiceControls();
    }
    function updateChoiceControls() {
      const count = draftIds.size;
      ui.selectionCount.textContent = `已選 ${count} / 8 組`;
      ui.applySelectionBtn.disabled = count !== PAIR_COUNT;
      ui.wordChoices.querySelectorAll("input").forEach((input) => {
        input.checked = draftIds.has(input.value);
        input.disabled = !input.checked && count >= PAIR_COUNT;
        input.closest("label").classList.toggle("is-selected", input.checked);
      });
    }
    ui.firstEightBtn.addEventListener("click", () => { draftIds = new Set(bank.slice(0, PAIR_COUNT).map((entry) => entry.id)); updateChoiceControls(); });
    ui.lastEightBtn.addEventListener("click", () => { draftIds = new Set(bank.slice(-PAIR_COUNT).map((entry) => entry.id)); updateChoiceControls(); });
    ui.applySelectionBtn.addEventListener("click", () => {
      if (draftIds.size !== PAIR_COUNT) return;
      selectedIds = bank.filter((entry) => draftIds.has(entry.id)).map((entry) => entry.id);
      hideDialog(ui.teacherDialog, false);
      resetRound(false);
      ui.startBtn.focus({ preventScroll: true });
    });

    // 成功提示音：只有兩個短音；絕不作為族語發音。
    function unlockSuccessAudio() {
      if (sfxUnavailable) return;
      const AudioContextClass = window.AudioContext || window.webkitAudioContext;
      if (!AudioContextClass) { markSfxUnavailable(); return; }
      try {
        if (!audioContext) {
          audioContext = new AudioContextClass();
          sfxMaster = audioContext.createGain();
          sfxMaster.gain.value = sfxMuted ? 0 : 1;
          sfxMaster.connect(audioContext.destination);
        }
        const promise = audioContext.resume();
        if (promise && typeof promise.catch === "function") promise.catch(() => { /* 遊戲不依賴音訊解鎖結果。 */ });
      } catch (_) { markSfxUnavailable(); }
    }
    function markSfxUnavailable() {
      sfxUnavailable = true;
      ui.muteLabel.textContent = "提示音：不可用";
      ui.muteBtn.disabled = true;
      ui.muteBtn.setAttribute("aria-label", "此瀏覽器無法使用成功提示音");
      ui.soundNotice.hidden = false;
      ui.soundNotice.textContent = "提示音無法使用，仍可正常遊戲。";
    }
    function playSuccessTone() {
      if (sfxMuted || !audioContext || !sfxMaster || sfxUnavailable) return;
      const session = state.session;
      const play = () => {
        if (sfxMuted || session !== state.session || audioContext.state !== "running") return;
        try {
          const now = audioContext.currentTime;
          [[660, 0], [880, .12]].forEach(([frequency, delay]) => {
            const oscillator = audioContext.createOscillator();
            const gain = audioContext.createGain();
            oscillator.type = "sine";
            oscillator.frequency.setValueAtTime(frequency, now + delay);
            gain.gain.setValueAtTime(0, now + delay);
            gain.gain.linearRampToValueAtTime(.055, now + delay + .015);
            gain.gain.exponentialRampToValueAtTime(.001, now + delay + .18);
            oscillator.connect(gain); gain.connect(sfxMaster);
            oscillator.start(now + delay); oscillator.stop(now + delay + .2);
            oscillator.onended = () => { oscillator.disconnect(); gain.disconnect(); };
          });
        } catch (_) { /* 提示音失敗不影響配對狀態。 */ }
      };
      if (audioContext.state === "suspended") audioContext.resume().then(play).catch(() => {}); else play();
    }
    ui.muteBtn.addEventListener("click", () => {
      sfxMuted = !sfxMuted;
      ui.muteBtn.setAttribute("aria-pressed", String(sfxMuted));
      ui.muteBtn.setAttribute("aria-label", sfxMuted ? "開啟成功提示音" : "靜音成功提示音");
      ui.muteLabel.textContent = sfxMuted ? "提示音：靜音" : "提示音：開";
      ui.soundWaves.style.display = sfxMuted ? "none" : "";
      ui.soundSlash.style.display = sfxMuted ? "" : "none";
      ui.soundSlash.removeAttribute("hidden");
      if (sfxMaster && audioContext) sfxMaster.gain.setValueAtTime(sfxMuted ? 0 : 1, audioContext.currentTime);
    });

    // 老師錄音獨立使用 HTMLAudioElement；僅按「聽發音」才載入、播放。
    // 版本號阻止重玩／切換詞彙之後，舊音檔的 Promise 或錯誤污染新畫面。
    function stopRecording() {
      recordingVersion += 1;
      if (recordingTimeout !== null) window.clearTimeout(recordingTimeout);
      recordingTimeout = null;
      if (activeRecording) {
        activeRecording.onerror = null; activeRecording.onended = null; activeRecording.onplaying = null;
        try { activeRecording.pause(); activeRecording.removeAttribute("src"); activeRecording.load(); } catch (_) {}
        activeRecording = null;
      }
    }
    function safeAudioSource(value) {
      const url = new URL(value, document.baseURI);
      if (["http:", "https:", "file:", "blob:"].includes(url.protocol)) return url.href;
      if (url.protocol === "data:" && /^data:audio\//i.test(value)) return value;
      throw new Error("不支援的音檔網址");
    }
    function playRecording() {
      const entry = state.feedbackEntry;
      if (state.phase !== "feedback" || !entry || !entry.audioUrl) return;
      stopRecording();
      const version = recordingVersion;
      ui.audioStatus.textContent = "";
      ui.pronounceLabel.textContent = "載入發音中…";
      let audio;
      const stillCurrent = () => version === recordingVersion && ui.feedbackDialog.open && state.feedbackEntry === entry;
      const fail = () => {
        if (!stillCurrent()) return;
        stopRecording();
        ui.pronounceLabel.textContent = "重試播放";
        ui.audioStatus.textContent = "音檔暫時無法播放";
      };
      try {
        audio = new Audio();
        activeRecording = audio;
        audio.preload = "none";
        audio.src = safeAudioSource(entry.audioUrl);
        audio.onerror = fail;
        audio.onplaying = () => {
          if (!stillCurrent()) return;
          if (recordingTimeout !== null) window.clearTimeout(recordingTimeout);
          recordingTimeout = null;
          ui.pronounceLabel.textContent = "重播發音";
          ui.audioStatus.textContent = "正在播放老師錄音，可再次按下重播。";
        };
        audio.onended = () => {
          if (!stillCurrent()) return;
          ui.pronounceLabel.textContent = "再聽一次";
          ui.audioStatus.textContent = "播放完畢。";
        };
        recordingTimeout = window.setTimeout(fail, 10000);
        const promise = audio.play();
        if (promise && typeof promise.catch === "function") promise.catch(fail);
      } catch (_) { fail(); }
    }

    ui.startBtn.addEventListener("click", startRound);
    ui.feedbackContinue.addEventListener("click", (event) => { if (event.detail <= 1) continueAfterFeedback(); });
    ui.pronounceBtn.addEventListener("click", playRecording);
    ui.playAgainBtn.addEventListener("click", () => resetRound(true));
    ui.reviewBoardBtn.addEventListener("click", () => { hideDialog(ui.resultDialog, false); ui.resultBtn.focus({ preventScroll: true }); });
    ui.resultBtn.addEventListener("click", showResults);
    ui.restartBtn.addEventListener("click", () => {
      if (state.phase === "playing") showDialog(ui.restartDialog, ui.cancelRestartBtn);
    });
    ui.cancelRestartBtn.addEventListener("click", () => hideDialog(ui.restartDialog));
    ui.confirmRestartBtn.addEventListener("click", () => resetRound(true));
    ui.teacherBtn.addEventListener("click", openTeacher);
    ui.closeTeacherBtn.addEventListener("click", () => hideDialog(ui.teacherDialog));
    ui.cancelTeacherBtn.addEventListener("click", () => hideDialog(ui.teacherDialog));
    ui.helpBtn.addEventListener("click", () => showDialog(ui.helpDialog, ui.helpTitle));
    ui.closeHelpBtn.addEventListener("click", () => hideDialog(ui.helpDialog));
    ui.helpDoneBtn.addEventListener("click", () => hideDialog(ui.helpDialog));
    ui.reduceMotionToggle.addEventListener("change", () => document.documentElement.classList.toggle("reduce-motion", ui.reduceMotionToggle.checked));
    window.addEventListener("pagehide", () => { stopRecording(); });
    document.addEventListener("visibilitychange", () => { if (document.hidden) stopRecording(); });
    init();
  })();

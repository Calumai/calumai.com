      item.textContent = state.phase === 'finished' ? '這一輪結束了，下方可以複習。' : '按「開始遊戲」，這裡就會出現題目。';
      ui.words.append(item);
      return;
    }
    for (const balloon of state.active.slice().sort((a, b) => b.progress - a.progress)) {
      const item = document.createElement('li');
      const word = document.createElement('span');
      word.className = 'typing-word';
      word.textContent = balloon.item.word;
      const meaning = document.createElement('span');
      meaning.className = 'typing-meaning';
      meaning.textContent = balloon.item.translation;
      item.append(word, meaning);
      ui.words.append(item);
    }
  }
  function stopLoop() {
    if (frameId !== null) cancelAnimationFrame(frameId);
    frameId = null;
    lastFrame = null;
  }
  function spawn() {
    if (state.nextIndex >= state.selected.length) return;
    const occupied = new Set(state.active.map(balloon => balloon.lane));
    const lane = occupied.has(0) ? 1 : 0;
    const index = state.nextIndex++;
    state.active.push({ uid: index, item: state.selected[index], lane, progress: 0 });
    state.spawnElapsed = 0;
    renderWords();
  }
  function start() {
    stopLoop();
    state = freshState();
    state.selected = shuffle(selectedPool()).slice(0, 8);
    state.mode = ui.still.checked ? 'practice' : ui.mode.value;
    state.speed = ui.speed.value;
    state.still = ui.still.checked;
    state.phase = 'running';
    state.selected.forEach(item => state.records.set(item.id, { outcome: 'pending', drops: 0 }));
    ui.input.value = '';
    ui.results.hidden = true;
    composing = false;
    spawn();
    updateControls();
    say(state.still ? '看著詞彙輸入，完成後按「送出」。不計時，可以慢慢來。' : '氣球開始落下了。輸入相同的族語，再按「送出」。');
    draw();
    scheduleFrame();
    ui.input.focus({ preventScroll: true });
    ui.form.scrollIntoView({ block: 'nearest', behavior: 'instant' });
  }
  function scheduleFrame() {
    if (state.phase === 'running' && !state.still && frameId === null) frameId = requestAnimationFrame(tick);
  }
  function tick(timestamp) {
    frameId = null;
    if (state.phase !== 'running' || state.still) return;
    const elapsed = lastFrame === null ? 0 : Math.min(timestamp - lastFrame, 100);
    lastFrame = timestamp;
    state.spawnElapsed += elapsed;
    for (const balloon of state.active.slice()) {
      if (state.phase !== 'running') break;
      balloon.progress += elapsed / duration[state.speed];
      if (balloon.progress >= 1) {
        state.misses++;
        state.records.get(balloon.item.id).drops++;
        if (state.mode === 'practice') {
          balloon.progress = 0;
          say(`${balloon.item.word} 回到上方了。繼續輸入，不扣心。`);
        } else {
          state.active = state.active.filter(other => other.uid !== balloon.uid);
          state.records.get(balloon.item.id).outcome = 'missed';
          state.resolved++;
          state.lives--;
          say(`漏接 ${balloon.item.word}，剩下 ${state.lives} 顆心。`, 'error');
          renderWords();
          if (state.lives === 0 || state.resolved === state.selected.length) finish();
        }
        updateControls();
      }
    }
    if (state.phase === 'running') {
      if ((!state.active.length || state.spawnElapsed >= 8000) && state.active.length < 2) spawn();
      draw();
      scheduleFrame();
    }
  }
  function submit() {
    if (state.phase !== 'running' || composing) return;
    const answer = ui.input.value;
    // 完全保留原拼寫：不轉小寫，也不移除任何空格。
    const matching = state.active.filter(balloon => balloon.item.word === answer)
      .sort((a, b) => b.progress - a.progress)[0];
    if (!matching) {
      state.wrongAnswers++;
      say(answer ? '再看一次拼寫。字母、大小寫和空格都要相同。' : '先在輸入框填入畫面上的族語。', 'error');
      ui.input.focus({ preventScroll: true });
      return;
    }
    state.active = state.active.filter(balloon => balloon.uid !== matching.uid);
    state.records.get(matching.item.id).outcome = 'correct';
    state.score += 10;
    state.resolved++;
    ui.input.value = '';
    say(`答對了！${matching.item.word} 是「${matching.item.translation}」，加 10 分。`);
    if (state.resolved === state.selected.length) { finish(); return; }
    if (!state.active.length) spawn();
    renderWords();
    updateControls();
    draw();
    ui.input.focus({ preventScroll: true });
  }
  function pause(auto = false) {
    if (state.phase !== 'running') return;
    stopLoop();
    state.phase = 'paused';
    updateControls();
    say(auto ? '已自動暫停。回來後按「繼續」。' : '已暫停，準備好後按「繼續」。');
    draw();
  }
  function resume() {
    if (state.phase !== 'paused') return;
    state.phase = 'running';
    lastFrame = null;
    updateControls();
    say('繼續練習，輸入與畫面相同的族語。');
    draw();
    scheduleFrame();
    ui.input.focus({ preventScroll: true });
  }
  function finish() {
    stopLoop();
    state.phase = 'finished';
    state.active = [];
    updateControls();
    renderWords();
    draw();
    const correct = state.score / 10;
    ui.resultTitle.textContent = state.lives === 0 ? '這一輪先到這裡' : '8 題練習完成';
    ui.summary.textContent = `答對 ${correct} 題，得到 ${state.score} 分。${state.lives === 0 ? '先看一次漏接的詞，再試一輪。' : '看看剛剛練過的數字，再選一組繼續。'}`;
    ui.review.replaceChildren();
    for (const item of state.selected) {
      const record = state.records.get(item.id);
      const row = document.createElement('li');
      const text = document.createElement('span');
      const word = document.createElement('strong');
      word.textContent = item.word;
      const meaning = document.createElement('small');
      meaning.textContent = item.translation;
      text.append(word, meaning);
      const outcome = document.createElement('span');
      outcome.className = 'typing-review-state';
      outcome.textContent = record.outcome === 'correct' ? '已完成' : record.outcome === 'missed' ? '漏接，再練一次' : '這輪還沒完成';
      row.append(text, outcome);
      ui.review.append(row);
    }
    ui.results.hidden = false;
    say(`這一輪結束，答對 ${correct} 題。下方可以複習。`);
    ui.resultTitle.focus({ preventScroll: true });
    ui.results.scrollIntoView({ block: 'nearest', behavior: 'instant' });
  }
  function resetToSettings() {
    stopLoop();
    state = freshState();
    ui.input.value = '';
    ui.results.hidden = true;
    updateControls();
    renderWords();
    draw();
    say('選好數字範圍與模式，再按「開始遊戲」。');
    ui.range.focus();
  }
  function roundRect(x, y, width, height, radius) {
    ctx.beginPath();
    ctx.moveTo(x + radius, y);
    ctx.arcTo(x + width, y, x + width, y + height, radius);
    ctx.arcTo(x + width, y + height, x, y + height, radius);
    ctx.arcTo(x, y + height, x, y, radius);
    ctx.arcTo(x, y, x + width, y, radius);
    ctx.closePath();
  }
  function drawBalloon(balloon, preview = false) {
    const wide = canvasWidth >= 500;
    const laneWidth = canvasWidth / 2;
    const centerX = state.active.length === 1 || preview ? canvasWidth / 2 : laneWidth * (balloon.lane + .5);
    const fontSize = wide ? 21 : 16;
    ctx.font = `700 ${fontSize}px "Microsoft JhengHei", system-ui, sans-serif`;
    const maxWidth = state.active.length > 1 ? laneWidth - 12 : canvasWidth - 30;
    const width = Math.min(Math.max(ctx.measureText(balloon.item.word).width + 28, 110), maxWidth);
    const height = 64;
    const travel = Math.max(canvasHeight - height - 42, 8);
    const y = 12 + (state.still || preview ? .2 : balloon.progress) * travel;
    ctx.strokeStyle = '#9dae89';
    ctx.lineWidth = 1.5;
    ctx.beginPath(); ctx.moveTo(centerX, y + height); ctx.lineTo(centerX, y + height + 16); ctx.stroke();
    roundRect(centerX - width / 2, y, width, height, 26);
    ctx.fillStyle = balloon.lane === 1 ? '#e8eadb' : '#dbe7cb'; ctx.fill();
    ctx.strokeStyle = '#98aa82'; ctx.stroke();
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillStyle = '#233d34';
    ctx.fillText(balloon.item.word, centerX, y + 24, width - 16);
    ctx.font = '14px "Microsoft JhengHei", system-ui, sans-serif';
    ctx.fillStyle = '#52634b'; ctx.fillText(balloon.item.translation, centerX, y + 47, width - 16);
  }
  function draw() {
    if (!ctx) return;
    ctx.clearRect(0, 0, canvasWidth, canvasHeight);
    ctx.fillStyle = '#f4f7ee'; ctx.fillRect(0, 0, canvasWidth, canvasHeight);
    ctx.fillStyle = '#e1e8d5'; ctx.fillRect(0, canvasHeight - 16, canvasWidth, 16);
    ctx.strokeStyle = '#b6c5a5'; ctx.beginPath(); ctx.moveTo(0, canvasHeight - 16); ctx.lineTo(canvasWidth, canvasHeight - 16); ctx.stroke();
    if (state.phase === 'ready') drawBalloon({ item: VOCAB[0], lane: 0, progress: .2 }, true);
    for (const balloon of state.active) drawBalloon(balloon);
    if (state.phase === 'paused' || state.phase === 'finished') {
      ctx.fillStyle = 'rgba(247,246,239,.9)'; ctx.fillRect(0, 0, canvasWidth, canvasHeight);
      ctx.fillStyle = '#294e40'; ctx.font = '700 24px "Microsoft JhengHei", system-ui, sans-serif';
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
      ctx.fillText(state.phase === 'paused' ? '已暫停' : '這一輪結束了', canvasWidth / 2, canvasHeight / 2);
    }
  }
  function resize() {
    const rect = ui.canvas.getBoundingClientRect();
    canvasWidth = Math.max(rect.width, 1);
    canvasHeight = Math.max(rect.height, 1);
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    ui.canvas.width = Math.round(canvasWidth * dpr);
    ui.canvas.height = Math.round(canvasHeight * dpr);
    if (ctx) ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    draw();
  }
  ui.form.addEventListener('submit', event => {
    event.preventDefault();
    if (!composing && performance.now() - compositionEndedAt > 80) submit();
  });
  ui.input.addEventListener('compositionstart', () => { composing = true; });
  ui.input.addEventListener('compositionend', () => { composing = false; compositionEndedAt = performance.now(); });
  ui.input.addEventListener('keydown', event => {
    if (event.key === 'Enter' && (composing || event.isComposing || event.keyCode === 229)) event.preventDefault();
  });
  ui.start.addEventListener('click', start);
  ui.again.addEventListener('click', start);
  ui.reset.addEventListener('click', start);
  ui.settings.addEventListener('click', resetToSettings);
  ui.pause.addEventListener('click', () => state.phase === 'paused' ? resume() : pause());
  ui.mode.addEventListener('change', () => { settingsNote(); updateControls(); });
  ui.still.addEventListener('change', () => { settingsNote(); updateControls(); });
  document.addEventListener('visibilitychange', () => { if (document.hidden) pause(true); });
  motionPreference.addEventListener('change', event => {
    if (!event.matches) return;
    ui.still.checked = true;
    if (state.phase === 'running' || state.phase === 'paused') {
      stopLoop(); state.still = true; state.mode = 'practice';
      updateControls(); draw();
      say('已改成不計時練習。氣球停在原地，可以慢慢輸入。');
    } else { settingsNote(); updateControls(); }
  });
  ui.still.checked = motionPreference.matches;
  new ResizeObserver(resize).observe(ui.canvas);
  updateControls();
  resize();
  // 只讀快照供檢查使用，不提供改分數、跳題或調整時間的方法。
  Object.defineProperty(window, 'gameDebug', { configurable: false, get() {
    return Object.freeze({ phase: state.phase, mode: state.mode, still: state.still,
      score: state.score, lives: state.lives, resolved: state.resolved, total: state.selected.length,
      nextIndex: state.nextIndex, loopScheduled: frameId !== null,
      active: Object.freeze(state.active.map(balloon => Object.freeze({ uid: balloon.uid, word: balloon.item.word,
        translation: balloon.item.translation, progress: balloon.progress }))) });
  } });
})();

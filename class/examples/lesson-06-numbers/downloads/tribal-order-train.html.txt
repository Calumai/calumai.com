    "exampleSentence": "",
    "audioUrl": ""
  },
  {
    "id": "n21",
    "word": "empusal kingal",
    "translation": "二十一",
    "exampleSentence": "",
    "audioUrl": ""
  },
  {
    "id": "n22",
    "word": "empusal dha",
    "translation": "二十二",
    "exampleSentence": "",
    "audioUrl": ""
  }
];
(() => {
  'use strict';

  // 只拆開題庫原有的空格，答案順序與原始詞彙完全相同。
  const bank = VOCAB.filter(entry => entry.word.includes(' ')).map(entry => ({
    id: entry.id,
    word: entry.word,
    translation: entry.translation,
    tokens: entry.word.split(' ').map((text, index) => ({ id: `${entry.id}-part-${index}`, text }))
  }));
  const $ = id => document.getElementById(id);
  const ui = {
    welcome: $('train-welcome'), round: $('train-round'), result: $('train-result'),
    question: $('train-question'), pool: $('train-pool'), carriages: $('train-carriages'),
    check: $('train-check'), reset: $('train-reset'), hint: $('train-hint'), next: $('train-next'),
    status: $('train-status'), hintText: $('train-hint-text'), stations: $('train-stations'),
    progress: $('train-progress'), completed: $('train-completed'), records: $('train-records')
  };
  let questions = [], questionIndex = 0, chosen = [], tokenOrder = [], records = [];
  let attempts = 0, hinted = false, solved = false, playing = false, lastChecked = null;

  function shuffle(items) {
    const result = [...items];
    for (let i = result.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [result[i], result[j]] = [result[j], result[i]];
    }
    return result;
  }

  function setStatus(text, kind = '') {
    ui.status.textContent = text;
    ui.status.dataset.kind = kind;
  }

  function renderStations() {
    ui.stations.replaceChildren();
    for (let i = 0; i < 6; i++) {
      const item = document.createElement('li');
      const done = i < records.length;
      item.textContent = done ? '✓' : String(i + 1);
      item.className = done ? 'is-done' : playing && i === questionIndex ? 'is-current' : '';
      item.setAttribute('aria-label', `第 ${i + 1} 題${done ? '已完成' : playing && i === questionIndex ? '，目前題目' : '尚未完成'}`);
      if (playing && i === questionIndex && !done) item.setAttribute('aria-current', 'step');
      ui.stations.append(item);
    }
    ui.completed.textContent = `${records.length} / 6`;
  }

  function focusAvailableToken() {
    const token = ui.pool.querySelector('button:not(:disabled)');
    (token || ui.check).focus();
  }

  function renderTrain() {
    const question = questions[questionIndex];
    ui.carriages.replaceChildren();
    question.tokens.forEach((unused, index) => {
      const carriage = document.createElement('div');
      carriage.className = `train-car${chosen[index] ? ' is-filled' : ''}${solved ? ' is-correct' : ''}`;
      const label = document.createElement('small');
      label.textContent = `第 ${index + 1} 節`;
      carriage.append(label);
      if (chosen[index]) {
        const token = question.tokens.find(part => part.id === chosen[index]);
        const button = document.createElement('button');
        button.type = 'button';
        button.textContent = token.text;
        button.dataset.tokenId = token.id;
        button.disabled = solved;
        button.setAttribute('aria-label', `第 ${index + 1} 節：${token.text}，點一下退回`);
        button.addEventListener('click', () => {
          if (!playing || solved) return;
          chosen.splice(index, 1);
          lastChecked = null;
          setStatus('詞塊已退回，點詞塊重新放上車。');
          renderTrain();
          focusAvailableToken();
        });
        carriage.append(button);
      } else {
        const empty = document.createElement('span');
        empty.textContent = '等詞塊上車';
        carriage.append(empty);
      }
      ui.carriages.append(carriage);
    });
    ui.pool.replaceChildren();
    tokenOrder.forEach(token => {
      const button = document.createElement('button');
      button.className = 'train-word';
      button.type = 'button';
      button.textContent = token.text;
      button.dataset.tokenId = token.id;
      button.disabled = solved || chosen.includes(token.id);
      button.setAttribute('aria-label', `${token.text}，放進下一節車廂`);
      button.addEventListener('click', () => {
        if (!playing || solved || chosen.includes(token.id)) return;
        chosen.push(token.id);
        lastChecked = null;
        setStatus(chosen.length === question.tokens.length ? '兩個詞都上車了，按「檢查答案」。' : '再點一個詞，放進下一節車廂。');
        renderTrain();
        focusAvailableToken();
      });
      ui.pool.append(button);
    });
    ui.check.disabled = solved;
    ui.reset.disabled = solved;
    ui.hint.disabled = solved;
  }

  function loadQuestion() {
    chosen = [];
    attempts = 0;
    hinted = false;
    solved = false;
    lastChecked = null;
    const question = questions[questionIndex];
    tokenOrder = shuffle(question.tokens);
    ui.question.textContent = question.translation;
    ui.question.dataset.questionId = question.id;
    ui.progress.textContent = `第 ${questionIndex + 1} 題`;
    ui.hintText.hidden = true;
    ui.hintText.textContent = '';
    ui.next.hidden = true;
    ui.next.textContent = questionIndex === questions.length - 1 ? '看本輪紀錄 →' : '下一題 →';
    setStatus('先點一個詞，放進第一節車廂。');
    renderStations();
    renderTrain();
    ui.question.focus({ preventScroll: true });
  }

  function startGame() {
    questions = ($('train-order').value === 'ordered' ? [...bank] : shuffle(bank)).slice(0, 6);
    records = [];
    questionIndex = 0;
    playing = true;
    ui.records.replaceChildren();
    ui.welcome.hidden = true;
    ui.result.hidden = true;
    ui.round.hidden = false;
    loadQuestion();
  }

  function finishGame() {
    playing = false;
    ui.round.hidden = true;
    ui.result.hidden = false;
    ui.progress.textContent = '全部到站';
    renderStations();
    const firstTry = records.filter(record => record.attempts === 1 && !record.hinted).length;
    $('train-summary').textContent = `完成 ${records.length} 題，其中 ${firstTry} 題不用提示就一次排對。下面可以再看一次完整詞組。`;
    ui.records.replaceChildren();
    records.forEach(record => {
      const row = document.createElement('li');
      const translation = document.createElement('strong');
      translation.textContent = record.translation;
      const word = document.createElement('span');
      word.textContent = record.word;
      const detail = document.createElement('small');
      detail.textContent = `${record.attempts === 1 ? '一次排對' : `調整後排對，檢查 ${record.attempts} 次`}${record.hinted ? '・看過提示' : ''}`;
      row.append(translation, word, detail);
      ui.records.append(row);
    });
    $('train-result-title').focus({ preventScroll: true });
  }

  ui.check.addEventListener('click', () => {
    if (!playing || solved) return;
    const question = questions[questionIndex];
    if (chosen.length !== question.tokens.length) {
      setStatus('還有詞塊沒上車，先把兩個詞都放進車廂。');
      return;
    }
    const arrangement = chosen.join('|');
    if (lastChecked === arrangement) {
      setStatus('這個排列還沒改。點車廂裡的詞退回，再換個順序試試。', 'retry');
      return;
    }
    attempts++;
    lastChecked = arrangement;
    if (chosen.every((id, index) => id === question.tokens[index].id)) {
      solved = true;
      records.push({ id: question.id, word: question.word, translation: question.translation, attempts, hinted });
      setStatus(`排列正確！${question.translation}：${question.word}。`, 'success');
      ui.hintText.hidden = true;
      ui.next.hidden = false;
      renderStations();
      renderTrain();
      ui.next.focus();
    } else {
      setStatus('順序還沒排對。點車廂裡的詞退回，換個順序再試一次。', 'retry');
    }
  });

  ui.reset.addEventListener('click', () => {
    if (!playing || solved) return;
    chosen = [];
    lastChecked = null;
    setStatus('車廂清空了，重新點選第一個詞。');
    renderTrain();
    focusAvailableToken();
  });

  ui.hint.addEventListener('click', () => {
    if (!playing || solved) return;
    hinted = true;
    ui.hintText.textContent = `第一節先放「${questions[questionIndex].tokens[0].text}」。`;
    ui.hintText.hidden = false;
    setStatus('提示在下面。先找到第一個詞，再把另一個詞接上。');
  });

  ui.next.addEventListener('click', () => {
    if (!playing || !solved) return;
    solved = false;
    questionIndex++;
    if (questionIndex >= questions.length) finishGame();
    else loadQuestion();
  });

  $('train-start').addEventListener('click', startGame);
  $('train-restart').addEventListener('click', startGame);
  bank.forEach(entry => {
    const item = document.createElement('li');
    const translation = document.createElement('span');
    translation.textContent = entry.translation;
    const word = document.createElement('span');
    word.textContent = entry.word;
    item.append(translation, word);
    $('train-bank-list').append(item);
  });
  renderStations();
})();

</script></body></html>

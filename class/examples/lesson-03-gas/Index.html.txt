  <main class="app">
    <section class="card" id="startView"><h1 id="title"></h1><p class="lead">這是單詞與句子測驗範例。完成後會將姓名、班級、座號與作答結果送到老師的 Google 試算表。</p><div class="grid"><div><label for="name">姓名或暱稱</label><input id="name" maxlength="40"></div><div><label for="className">班級</label><input id="className" maxlength="30" placeholder="七年一班"></div><div><label for="seat">座號</label><input id="seat" maxlength="12" inputmode="numeric"></div></div><div class="actions"><button id="startBtn">開始測驗</button></div></section>
    <section class="card hidden" id="quizView"><div class="meta"><span id="count"></span><span id="scoreNow"></span></div><div class="progress"><i id="bar"></i></div><p class="question-type" id="qtype"></p><h2 id="question"></h2><div class="options" id="options"></div><div class="feedback hidden" id="feedback"></div><div class="actions"><button class="hidden" id="nextBtn">下一題</button></div></section>
    <section class="card hidden" id="resultView"><h2>完成測驗</h2><div class="score" id="finalScore"></div><p class="lead" id="resultText"></p><div class="actions"><button id="sendBtn">送出成績</button><button class="secondary" id="restartBtn">重新挑戰</button></div><div class="sendState hidden" id="sendState" role="status" aria-live="polite"></div></section>

  <script>
    // 只要替換 title 與 QUIZ_DATA，就能改成自己的單詞或句子測驗。
    // type、question、options、answer、explanation 這五個欄位請保留。
    const QUIZ_INFO={title:'單詞與句子小測驗'};
    const QUIZ_DATA=[
      {type:'單詞題',question:'「book」最適合翻成哪一個中文意思？',options:['書','桌子','老師','朋友'],answer:0,explanation:'book 是「書」。請依自己的教材替換單詞與選項。'},
      {type:'單詞題',question:'「happy」最接近哪一個意思？',options:['疲累','開心','安靜','寒冷'],answer:1,explanation:'happy 是「開心」。選項可以改成你的族語、英語或其他課堂詞彙。'},
      {type:'句子題',question:'「I like reading books.」最適合的意思是？',options:['我正在買書。','我喜歡讀書。','我把書借給你。','我今天沒有帶書。'],answer:1,explanation:'先讓學生練習理解完整句子，再依程度加入情境或語法提示。'},
      {type:'句子題',question:'要表達「我明天會去學校」，哪一句最合適？',options:['I went to school yesterday.','I am at school now.','I will go to school tomorrow.','I like my school.'],answer:2,explanation:'will go 表示未來要去，tomorrow 表示明天。'},
      {type:'句子題',question:'看到「Thank you for your help.」時，最適合的回應是？',options:['You are welcome.','Good night.','I am hungry.','See you yesterday.'],answer:0,explanation:'You are welcome. 是回應感謝的常用句子。'}
    ];
    const $=id=>document.getElementById(id);const views=['startView','quizView','resultView'];let state={index:0,score:0,locked:false,answers:[],sent:false};$('title').textContent=QUIZ_INFO.title;
    function show(id){views.forEach(v=>$(v).classList.toggle('hidden',v!==id));window.scrollTo({top:0,behavior:'smooth'})}
    function start(){if(!$('name').value.trim()||!$('className').value.trim()||!$('seat').value.trim()){alert('請填寫姓名、班級與座號。');return}state={index:0,score:0,locked:false,answers:[],sent:false};show('quizView');render()}
    function render(){const q=QUIZ_DATA[state.index];state.locked=false;$('count').textContent=`第 ${state.index+1} / ${QUIZ_DATA.length} 題`;$('scoreNow').textContent=`目前 ${state.score} 分`;$('bar').style.width=`${(state.index+1)/QUIZ_DATA.length*100}%`;$('qtype').textContent=q.type||'測驗題';$('question').textContent=q.question;$('feedback').classList.add('hidden');$('nextBtn').classList.add('hidden');$('options').innerHTML='';q.options.forEach((text,i)=>{const b=document.createElement('button');b.className='option';b.textContent=`${String.fromCharCode(65+i)}. ${text}`;b.onclick=()=>answer(i,b);$('options').appendChild(b)})}
    function answer(choice,button){if(state.locked)return;state.locked=true;const q=QUIZ_DATA[state.index];const ok=choice===q.answer;if(ok)state.score++;state.answers.push({type:q.type||'測驗題',question:q.question,choice:q.options[choice],correct:q.options[q.answer],isCorrect:ok});[...$('options').children].forEach((b,i)=>{b.disabled=true;if(i===q.answer)b.classList.add('correct')});if(!ok)button.classList.add('wrong');$('feedback').textContent=`${ok?'答對了！':'再看一次。'} ${q.explanation}`;$('feedback').classList.remove('hidden');$('scoreNow').textContent=`目前 ${state.score} 分`;$('nextBtn').textContent=state.index===QUIZ_DATA.length-1?'查看結果':'下一題';$('nextBtn').classList.remove('hidden')}
    function next(){if(state.index<QUIZ_DATA.length-1){state.index++;render()}else finish()}
    function finish(){const percent=Math.round(state.score/QUIZ_DATA.length*100);$('finalScore').textContent=`${percent}%`;$('resultText').textContent=`答對 ${state.score} / ${QUIZ_DATA.length} 題。確認資料後送出給老師。`;$('sendBtn').disabled=false;$('sendBtn').textContent='送出成績';$('sendState').classList.add('hidden');show('resultView')}
    function send(){if(state.sent)return;const button=$('sendBtn');button.disabled=true;button.textContent='傳送中…';const payload={name:$('name').value.trim(),className:$('className').value.trim(),seat:$('seat').value.trim(),quizTitle:QUIZ_INFO.title,score:state.score,total:QUIZ_DATA.length,answers:state.answers};const success=result=>{state.sent=true;button.textContent='已送出';$('sendState').textContent=result&&result.message?result.message:'成績已送出';$('sendState').classList.remove('hidden')};const failure=error=>{button.disabled=false;button.textContent='重新送出';$('sendState').textContent=`傳送失敗：${error&&error.message?error.message:'請稍後再試'}`;$('sendState').classList.remove('hidden')};if(window.google&&google.script&&google.script.run){google.script.run.withSuccessHandler(success).withFailureHandler(failure).saveQuizResult(payload)}else{setTimeout(()=>success({message:'公開預覽模式：流程成功，但不會寫入試算表。'}),500)}}
    $('startBtn').onclick=start;$('nextBtn').onclick=next;$('sendBtn').onclick=send;$('restartBtn').onclick=()=>show('startView');
  </script>

    <section class="card hidden" id="teacherLoginView"><h2>教師題庫管理</h2><p class="lead">輸入你在 <strong>Code.gs</strong> 設定的教師 PIN，才能新增、修改或刪除題目。</p><label for="teacherPin">教師 PIN</label><input id="teacherPin" type="password" autocomplete="current-password"><div class="actions"><button id="teacherLoginBtn">進入題庫管理</button><button class="secondary" data-back>回學生測驗</button></div><div class="send-state hidden" id="teacherLoginState" role="status" aria-live="polite"></div></section>

    <section class="card hidden" id="teacherView"><div class="editor-head"><div><h2>題庫管理</h2><p class="lead">可自由新增單詞或句子題；每題至少要有 2 個選項，並勾選 1 個正確答案。</p></div><button class="secondary" data-back>回學生測驗</button></div><div class="notice small">題目會儲存在同一份 Google Sheet 的「題庫」工作表。學生只能作答，無法進入這個管理區。</div><label for="teacherTitle">測驗名稱</label><input id="teacherTitle" maxlength="80"><div id="questionEditors"></div><div class="actions"><button class="secondary" id="addQuestionBtn">＋ 新增一題</button><button id="saveQuestionsBtn">儲存題庫</button></div><div class="send-state hidden" id="teacherSaveState" role="status" aria-live="polite"></div></section>

  <script>
    const $ = id => document.getElementById(id);
    const viewIds = ['startView','quizView','resultView','teacherLoginView','teacherView'];
    const defaultQuiz = {title:'單詞與句子小測驗',questions:[
      {type:'單詞題',question:'「book」最適合翻成哪一個中文意思？',options:['書','桌子','老師','朋友'],answer:0,explanation:'book 是「書」。'},
      {type:'句子題',question:'「I like reading books.」最適合的意思是？',options:['我正在買書。','我喜歡讀書。','我把書借給你。','我今天沒有帶書。'],answer:1,explanation:'請依課堂教材改成自己的句子題。'}
    ]};
    const isGas = () => window.google && google.script && google.script.run;
    let quiz = JSON.parse(sessionStorage.getItem('teacherQuiz') || JSON.stringify(defaultQuiz));
    let teacherPin = '';
    let state = {index:0,score:0,locked:false,answers:[],sent:false};

    function show(id){viewIds.forEach(view => $(view).classList.toggle('hidden',view !== id));window.scrollTo({top:0,behavior:'smooth'})}
    function setMessage(id,text){$(id).textContent=text;$(id).classList.remove('hidden')}
    function clearMessage(id){$(id).classList.add('hidden');$(id).textContent=''}
    function callGas(method,args){return new Promise((resolve,reject)=>{let runner=google.script.run.withSuccessHandler(resolve).withFailureHandler(reject);runner[method].apply(runner,args)})}
    async function api(method,...args){
      if(isGas()) return callGas(method,args);
      if(method === 'getQuizData') return quiz;
      if(method === 'getTeacherQuiz'){if(!args[0]) throw new Error('公開預覽請輸入任意 PIN。');return quiz}
      if(method === 'saveTeacherQuiz'){quiz=args[1];sessionStorage.setItem('teacherQuiz',JSON.stringify(quiz));return {success:true,message:'公開預覽已暫存題庫；正式部署後會寫入 Google Sheet。'}}
      if(method === 'saveQuizResult') return {success:true,message:'公開預覽模式：流程成功，但不會寫入試算表。'};
    }
    async function loadQuiz(){quiz=await api('getQuizData');$('title').textContent=quiz.title}

    function start(){if(!$('name').value.trim()||!$('className').value.trim()||!$('seat').value.trim()){alert('請填寫姓名、班級與座號。');return}if(!quiz.questions.length){alert('題庫沒有題目，請先請老師新增。');return}state={index:0,score:0,locked:false,answers:[],sent:false};show('quizView');renderQuestion()}
    function renderQuestion(){const q=quiz.questions[state.index];state.locked=false;$('count').textContent='第 '+(state.index+1)+' / '+quiz.questions.length+' 題';$('scoreNow').textContent='目前 '+state.score+' 分';$('bar').style.width=((state.index+1)/quiz.questions.length*100)+'%';$('qtype').textContent=q.type||'測驗題';$('question').textContent=q.question;$('feedback').classList.add('hidden');$('nextBtn').classList.add('hidden');$('options').innerHTML='';q.options.forEach((text,i)=>{const b=document.createElement('button');b.className='option';b.textContent=String.fromCharCode(65+i)+'. '+text;b.onclick=()=>answer(i,b);$('options').appendChild(b)})}
    function answer(choice,button){if(state.locked)return;state.locked=true;const q=quiz.questions[state.index],ok=choice===q.answer;if(ok)state.score++;state.answers.push({type:q.type||'測驗題',question:q.question,choice:q.options[choice],correct:q.options[q.answer],isCorrect:ok});[...$('options').children].forEach((b,i)=>{b.disabled=true;if(i===q.answer)b.classList.add('correct')});if(!ok)button.classList.add('wrong');$('feedback').textContent=(ok?'答對了！':'再看一次。')+' '+q.explanation;$('feedback').classList.remove('hidden');$('scoreNow').textContent='目前 '+state.score+' 分';$('nextBtn').textContent=state.index===quiz.questions.length-1?'查看結果':'下一題';$('nextBtn').classList.remove('hidden')}
    function next(){if(state.index<quiz.questions.length-1){state.index++;renderQuestion()}else finish()}
    function finish(){const percent=Math.round(state.score/quiz.questions.length*100);$('finalScore').textContent=percent+'%';$('resultText').textContent='答對 '+state.score+' / '+quiz.questions.length+' 題。確認資料後送出給老師。';$('sendBtn').disabled=false;$('sendBtn').textContent='送出成績';clearMessage('sendState');show('resultView')}
    async function send(){if(state.sent)return;const button=$('sendBtn');button.disabled=true;button.textContent='傳送中…';try{const result=await api('saveQuizResult',{name:$('name').value.trim(),className:$('className').value.trim(),seat:$('seat').value.trim(),quizTitle:quiz.title,score:state.score,total:quiz.questions.length,answers:state.answers});state.sent=true;button.textContent='已送出';setMessage('sendState',result.message||'成績已送出。')}catch(error){button.disabled=false;button.textContent='重新送出';setMessage('sendState','傳送失敗：'+(error.message||error))}}

    function addQuestion(question) {
      const q = question || {type:'單詞題',question:'',options:['','','',''],answer:0,explanation:''};
      const card = document.createElement('fieldset');
      const radioName = 'correct-' + Date.now() + '-' + Math.random();
      card.className = 'editor-card question-editor';
      card.innerHTML = '<div class="editor-head"><strong>題目</strong><button type="button" class="danger remove-question">刪除這題</button></div><div class="teacher-grid"><div><label>題型</label><select class="edit-type"><option>單詞題</option><option>句子題</option></select></div><div><label>正確答案</label><span class="muted small">在選項左側勾選</span></div></div><label>題目內容</label><textarea class="edit-question" maxlength="300"></textarea><div class="edit-options"></div><label>答題後解析</label><textarea class="edit-explanation" maxlength="500"></textarea>';
      card.querySelector('.edit-type').value = q.type || '單詞題';
      card.querySelector('.edit-question').value = q.question || '';
      card.querySelector('.edit-explanation').value = q.explanation || '';
      const options = card.querySelector('.edit-options');
      for (let i = 0; i < 4; i++) {
        const row = document.createElement('label');
        row.className = 'choice-row';
        row.innerHTML = '<input class="edit-correct" type="radio" name="' + radioName + '" value="' + i + '"><input class="edit-option" type="text" maxlength="120" placeholder="選項 ' + String.fromCharCode(65 + i) + '">';
        row.querySelector('.edit-option').value = (q.options || [])[i] || '';
        row.querySelector('.edit-correct').checked = Number(q.answer) === i;
        options.appendChild(row);
      }
      card.querySelector('.remove-question').onclick = () => {
        if (document.querySelectorAll('.question-editor').length === 1) {
          alert('至少保留一題。');
          return;
        }
        card.remove();
      };
      $('questionEditors').appendChild(card);
    }
    function readEditor(){const questions=[...document.querySelectorAll('.question-editor')].map(card=>{const options=[...card.querySelectorAll('.edit-option')].map(input=>input.value.trim()).filter(Boolean);const selected=card.querySelector('.edit-correct:checked');const selectedOriginal=selected?Number(selected.value):-1;const originalOptions=[...card.querySelectorAll('.edit-option')].map(input=>input.value.trim());const answer=originalOptions.slice(0,selectedOriginal).filter(Boolean).length;if(!card.querySelector('.edit-question').value.trim())throw new Error('每一題都要填題目內容。');if(options.length<2)throw new Error('每題至少要填 2 個選項。');if(selectedOriginal<0||!originalOptions[selectedOriginal])throw new Error('每一題都要勾選一個有內容的正確答案。');return {type:card.querySelector('.edit-type').value,question:card.querySelector('.edit-question').value.trim(),options,answer:answer,explanation:card.querySelector('.edit-explanation').value.trim()}});return {title:$('teacherTitle').value.trim(),questions}}
    function renderTeacher(data){$('teacherTitle').value=data.title||'';$('questionEditors').innerHTML='';data.questions.forEach(addQuestion);show('teacherView')}
    async function loginTeacher(){clearMessage('teacherLoginState');teacherPin=$('teacherPin').value;try{renderTeacher(await api('getTeacherQuiz',teacherPin))}catch(error){setMessage('teacherLoginState',error.message||String(error))}}
    async function saveTeacher(){clearMessage('teacherSaveState');let data;try{data=readEditor();if(!data.title)throw new Error('請填寫測驗名稱。')}catch(error){setMessage('teacherSaveState',error.message);return}const button=$('saveQuestionsBtn');button.disabled=true;button.textContent='儲存中…';try{const result=await api('saveTeacherQuiz',teacherPin,data);quiz=data;$('title').textContent=data.title;setMessage('teacherSaveState',result.message||'題庫已儲存。')}catch(error){setMessage('teacherSaveState',error.message||String(error))}finally{button.disabled=false;button.textContent='儲存題庫'}}
    async function backToStudent(){try{await loadQuiz()}catch(error){alert(error.message||String(error))}show('startView')}

    $('startBtn').onclick=start;$('nextBtn').onclick=next;$('sendBtn').onclick=send;$('restartBtn').onclick=()=>show('startView');$('teacherBtn').onclick=()=>{clearMessage('teacherLoginState');show('teacherLoginView')};$('teacherLoginBtn').onclick=loginTeacher;$('addQuestionBtn').onclick=()=>addQuestion();$('saveQuestionsBtn').onclick=saveTeacher;document.querySelectorAll('[data-back]').forEach(button=>button.onclick=backToStudent);loadQuiz().catch(error=>alert(error.message||String(error)));
  </script>
